const FINDINGS = {
  type: 'object', required: ['findings', 'gate_green', 'summary'],
  properties: {
    gate_green: { type: 'boolean' }, summary: { type: 'string' },
    findings: { type: 'array', items: { type: 'object', required: ['severity', 'title', 'location', 'description', 'fix'],
      properties: { severity: { type: 'string', enum: ['high', 'medium', 'low'] }, title: { type: 'string' }, location: { type: 'string' }, description: { type: 'string' }, fix: { type: 'string' } } } },
  },
}

async function reviewFix(label, phaseName, scope, maxRounds = 2) {
  let round = 0, blocking = []
  while (round < maxRounds) {
    round++
    const reviews = (await parallel([
      () => agent(`${CONTEXT}\n\nYou are the hostile security reviewer (${ROOT}/.claude/agents/security-reviewer.md). Review the current uncommitted diff ('git status', 'git diff', read every new file) for this scope: ${scope}. Check it against docs/PLAN.md §B.3, §B.6 and §B.8 (esp. §B.8.16) line by line where applicable, run '${BUILD} cargo test -p <crates touched>' yourself, and report findings with severity (high = must fix before tag, medium = fix now, low = note). Look for: missing or wrong cryptographic binding, replay/reflection, oracle/enumeration, secrets in logs/Debug/errors, non-constant-time compares, unbounded allocation from untrusted input, panics on malformed input, missing tests for security-relevant behaviour, unsafe or forbidden dependencies.`,
        { label: `${label}:review-security#${round}`, phase: phaseName, schema: FINDINGS, effort: 'max' }),
      () => agent(`${CONTEXT}\n\nReview the current uncommitted diff ('git status', 'git diff', read every new file) for CORRECTNESS and COMPLETENESS for this scope: ${scope}. Compare every implemented struct/message/state/constant with the plan text (docs/PLAN.md §B.3/§B.8 sections named in the scope) and with the verified API notes in docs/research/crypto.md / arti.md; check the tests actually exercise the specified behaviour (not just happy paths); check docs (DEPENDENCIES.md rows for new crates, VERIFICATION.md rows, decisions.md) are updated; run '${BUILD} cargo xtask verify' and report whether it is green. Report findings with severity.`,
        { label: `${label}:review-correctness#${round}`, phase: phaseName, schema: FINDINGS, effort: 'max' }),
    ])).filter(Boolean)
    const all = reviews.flatMap(r => r.findings)
    blocking = all.filter(f => f.severity !== 'low')
    const gate = reviews.every(r => r.gate_green)
    log(`${label} review round ${round}: ${blocking.length} blocking, ${all.length - blocking.length} low, gate_green=${gate}`)
    if (blocking.length === 0 && gate) break
    await agent(`${CONTEXT}\n\nApply these review findings for scope "${scope}" (read the files first; keep style; add tests where a finding is about behaviour). If a finding is wrong, explain why instead of changing code. Finish with '${BUILD} cargo xtask verify' green. Return what you changed.\n\n` + all.map(f => `- [${f.severity}] ${f.title} @ ${f.location}: ${f.description}\n  FIX: ${f.fix}`).join('\n'),
      { label: `${label}:fix#${round}`, phase: phaseName, effort: 'max' })
  }
  return { rounds: round, remainingBlocking: blocking }
}
